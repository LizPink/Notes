假设有`users`用户注册信息和`user_events`用户行为两张表。下面这个方法这是留存分析的一种常用算法，这个算法的好处是拓展性强，可以依据现有代码生成d1/d3/d7的留存率数据。
```SQL
WITH d1_label AS
(
  SELECT
    u.register_time::DATE AS register_date,
    u.user_id,
    EXISTS(
      SELECT 1
      FROM
        user_events AS e
      WHERE
        e.user_id = u.user_id
        AND
        e.event_time::DATE = u.register_time::DATE + INTERVAL '1 Day'
    )::INT AS d1_flag
  FROM
    users AS u
)

SELECT
  register_date,
  COUNT(DISTINCT user_id) AS num_registers,
  SUM(d1_flag) AS num_d1_retained,
  ROUND(
  	SUM(d1_flag)::NUMERIC / NULLIF(COUNT(DISTINCT user_id),0)
  ,2) AS d1_retained_rate
FROM
  d1_label
GROUP BY
  register_date
```
