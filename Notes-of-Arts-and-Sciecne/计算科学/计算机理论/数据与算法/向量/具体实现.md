# 向量实现
## 静态向量
```cpp
/* 向量 */
#include <iostream>
#include <algorithm>
using Rank = int;
constexpr int DEFAULT_CAPACITY = 3;

template <typename T>
class Vector{
    private:
        Rank _size;         // 当前规模
        int _capacity;      // 最大容量
        T* _elem;           // 数据区域
    public:
        // 类构造函数
        Vector(int c=DEFAULT_CAPACITY);
        Vector(const T* A, Rank low, Rank high);
        Vector(const Vector<T> &V, Rank low, Rank high);
        void copyFrom(const T* A, Rank lo, Rank hi);
        // 类析构函数
        ~Vector();
};

int  main()
{
    int i = 1;
    std::cout << (i <<= 1);
    return 0;
}

template <typename T>
Vector<T>::Vector(int c)
{
    _elem = new T[_capacity = c];
    _size = 0;
}
template <typename T>
Vector<T>::Vector(const Vector<T> &V, Rank low, Rank high)
{
    copyFrom(V._elem, low, high);
}
template <typename T>
Vector<T>::Vector(const T* A, Rank low, Rank high)
{
    copyFrom(A, low, high);
}
template <typename T>
void Vector<T>::copyFrom(const T* A, Rank lo, Rank hi)
{
    _capacity = 2*(hi-lo);
    _elem = new T[_capacity];
    _size = 0;
    while(lo < hi)
    {
        _elem[_size] = A[lo];
        _size += 1;
        lo += 1;
    }
}
template <typename T>
Vector<T>::~Vector()
{
    delete[] _elem;
}
```

## 动态向量
常见的扩容策略有递增扩容和加倍扩容。递增扩容的空间利用率较高，但是时间复杂度较高；加倍扩容的空间利用率较低，但是时间复杂度最低。
```cpp
template <typename T>
/* 倍增扩容功能 */
class Vector{
	...
	
	// 向量扩容函数
	void expand();
}

template <typename T>
void Vector<T>::expand()
{
    if (_size < _capacity) return;                      	// 尚未满员时不必扩容
    _capacity = std::max(_capacity, DEFAULT_CAPACITY);  	// 不低于最小容量
	T* oldElem = _elem; _elem = new T[_capacity <<= 1];   	// 容量加倍
	
    for (int i=0; i<_size; i++)							// 复制内容
    {
        _elem[i] = oldElem[i];
    }
    delete[] oldElem;
}
```

# 元素增加
经典的元素插入算法是在插入位置后，从后向前依次将元素后移，最后在插入位置写入元素。
```cpp
template <typename T>
class Vector{
	...
public:
	Rank insert(Rank r, const T& e);
};

template <typename T>
Rank Vector<T>::insert(Rank r, const T& e)
{
	expand();	// 如有必要则扩容
	for (int i = _size; i>r; i--)	// 自后向前
	{
		_elem[i] = _elem[i-1];		// 后继元素依次后移一个单位
	}
	_elem[r] = e; _size += 1;		// 置入新元素，更新容量
	return r;
}
```

# 元素删除

```cpp
template <typename T>
class Vector{
	...
public:
	int remove(Rank lo, Rank hi);
};

template <typename T>	// 删除区间[lo,hi)
int Vector<T>::remove(Rank lo, Rank hi)
{
	if (lo == hi) {return 0;}
	while (hi < _size) {_elem[lo++] = _elem[hi++];}
	
	_size = lo;			// 更新规模
	return hi-lo;		// 返回被删除元素的个数
}
```

# 元素查询
## 无序向量
### 循序查找
首先我们介绍一个逆序查询算法，该算法的时间复杂度为O(1)-O(n)阶。
> [!video-medium]
> ![[无序向量-逆序查找.mp4]]
```cpp
template <typename T>
class Vector{
...
public:
	Rank search(const T& e, Rank lo, Rank hi) const;
};

template <typename T>	// 0 <= lo < hi <= _size
Rank Vector<T>::find(const T& e, Rank lo, Rank hi) const	// 逆向查找：在命中多个元素时返回秩最大者
{
	while ((lo<hi--) && (e!=_elem[hi])){}					// 逆向查找
	return hi;			// 返回最终停止的索引：hi<lo意味着查找失败，否则hi即命中元素的秩。
}
```
## 有序向量
### 二分查找
#### 查找规则
在这里我们约定使用"左闭右开"的区间规则，在这个体系下区间的中间元素会是"中间偏右"的分布。

有序向量的二分查找总体上分为"左边界"和"右边界"两种模式，前者返回"目标元素左边界元素"的位置，后者返回"目标元素右边界元素"的位置，两种模式对于区间分割的差异仅在中点元素上：如果中间元素等于查找元素，在第一个模式下再次考察它左边的元素，在第二个模式下再次考察他右边的元素。

在左闭右开体系下，采用空列表终止条件时：当目标元素存在时，左边界是相同元素的左侧闭位置，右边界是相同元素的右侧开位置；当目标元素不存在时，左边界和右边界都指向距离目标元素最近的右侧开位置。

第一个算法是精确查找。
![[VectorBinarySearchB 1.mp4]]
```cpp
/* Binary Search (b) */
template <typename T>
class Vector{
...
public:
	Rank binary_search(const T& e, Rank lo, Rank hi) const;
};

template <typename T>				// 右边界查找：返回目标元素的右侧开位置
Rank Vector<T>::binary_search(const T& e, Rank lo, Rank hi) const
{
	Rank begin = lo, end = hi;
	while(lo < hi)					// 终止条件：形成空列表
	{
		Rank mid = (lo+hi) >> 1;	// 计算中间位置
		
		if (e < _elem[mid])
			{hi = mid;}
		else
			{lo = mid+1;}
	}
	Rank pos = lo;					// pos = lo = hi
	return (pos>begin && _elem[pos-1]==e) ? pos-1 : -1;	// 处理左右边界条件，返回命中元素的秩或者-1
}
```
第二个算法是位置查找。
```cpp
/* Binary Search (c) */
template <typename T>
class Vector{
...
public:
	Rank binary_search(const T& e, Rank lo, Rank hi) const;
};

template <typename T>				// 右边界查找：返回距离目标元素最近的右侧开位置
Rank Vector<T>::binary_search(const T& e, Rank lo, Rank hi) const
{
	Rank begin = lo, end = hi;
	while(lo < hi)					// 终止条件：形成空列表
	{
		Rank mid = (lo+hi) >> 1;	// 计算中间位置
		
		if (e < _elem[mid])
			{hi = mid;}
		else
			{lo = mid+1;}
	}
	Rank pos = lo;					// pos = lo = hi
	return pos
}
```
这两个算法的接口都是统一的，并且两个算法的终止条件都是形成一个空向量。
# 元素去重
## 无序向量
>[!video-medium]
>![[无序向量-去重.mp4]]
```cpp
template <typename T>
class Vector{
	...
public:
	int unique();
};

template <typename T>
int Vector<T>::unique()
{
    int oldSize = _size;    // 记录原规模
    Rank i = 1;             // 从_elem[1]开始
    while(i < _size)        // 自前向后逐一考察各元素
    {
        (find(_elem[i], 0, i) < 0)  // 在前缀中寻找雷同者
        ? i++;                      // 若无雷同者，则往后继续查
        : remove(i,i+1);            // 若有雷同者，则删除雷同者
    }
    return oldSize - _size;         // 返回向量被删除的个数
}
```
## 有序向量

```cpp
template <typename T>
class Vector{
	...
public:
	int unique();
};

int Vector<T>::unique()
{
    Rank i=0, j=0;
    while(++j < _size)  // 逐一扫描，直至扫描所有元素
    {
        // 跳过雷同者；发现不同元素时，前移至新元素右侧
        if (_elem[i] != _elem[j])
        {
            _elem[++i] = _elem[j];
        }
    }
    _size = ++i;
    return j-i;         // 返回被删除的元素总数
}
```
