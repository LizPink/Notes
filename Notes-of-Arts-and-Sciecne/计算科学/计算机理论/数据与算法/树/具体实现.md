# 二叉树的定义
```cpp
#include <iostream>

template <typename T>
class BinNode{
using BinNodePosition = BinNode<T>*;
private:
    BinNodePosition parent=nullptr, lchild=nullptr, rchild=nullptr;  // 父亲与孩子节点
    T data;                 // 数据区域
    int height, size;       // 高度与子树规模
public:
    BinNodePosition insertAsLC(const T&);
    BinNodePosition insertAsRC(const T&);
    BinNodePosition successor();      // 中序遍历当前节点的直接后继
    template <typename VST>
    void travelLevel(VST&);
    template <typename VST>
    void travelPre(VST&);
    template <typename VST>
    void trevelIn(VST&);
    template <typename VST>
    void trevelPost(VST&);
};
```
