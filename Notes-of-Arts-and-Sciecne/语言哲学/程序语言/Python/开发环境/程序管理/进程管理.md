```python
import multiprocessing

def work(*args):
	pass

if __name__ == "main":
	subprocesses = []
	## 创建并运行子进程
	for i in range(M):
		subprocess = multiprocessing.Process(target=work, args=(arg1, ..., argN))
		subprocesses.append(subprocess)
		subprocess.start()
	## 进程间通信
	...
	## 等待子进程的结束
	for p in subprocesses:
		p.join()
```
